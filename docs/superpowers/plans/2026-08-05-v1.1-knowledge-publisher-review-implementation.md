```python
def test_official_approved_seed_publishes_and_is_queryable(tmp_path: Path) -> None:
    bundle = load_knowledge_seed(OFFICIAL_SEED_DIRECTORY)
    database_path = tmp_path / "official-published.sqlite3"
    database = KnowledgeDatabase(database_path)
    initialize_knowledge_schema(database)
    importer = KnowledgeSeedImporter(database)

    try:
        first = importer.import_seed(bundle)
        second = importer.import_seed(bundle)
        published = KnowledgeVersionPublisher(database).publish(first.version_id)
        repository = SqliteKnowledgeRepository(database)
        version = repository.get_published_version("2026.08.1")

        assert first.created is True
        assert second.idempotent is True
        assert published.status == "published"
        assert published.content_hash == EXPECTED_APPROVED_CONTENT_HASH
        assert version is not None
        assert version.version == "2026.08.1"
        assert version.content_hash == EXPECTED_APPROVED_CONTENT_HASH

        with pytest.raises(KnowledgeSeedError) as exc_info:
            KnowledgeVersionPublisher(database).publish(first.version_id)
        assert exc_info.value.code is ErrorCode.KNOWLEDGE_PUBLISHED_IMMUTABLE
    finally:
        database.dispose()
```
